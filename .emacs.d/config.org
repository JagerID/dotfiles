  ;; внешних программ (psql, shell) «сырыми байтами»; кириллица в нём ломает разбор JSON.

  (defun make-missing-dirs ()
    (when buffer-file-name
      (let ((dir (file-name-directory buffer-file-name)))
        (when (and dir (not (file-exists-p dir)))
          (make-directory dir t)))))

  (defun trim-whitespace ()
    (add-hook 'before-save-hook #'delete-trailing-whitespace nil t))

  (defun my/compile-config ()
    "Байт- и нативная компиляция lisp/ после tangle: Emacs стартует быстрее.
Запускать вручную после каждого tangle. Если что-то сломалось — удалите lisp/*.elc."
    (interactive)

      (byte-recompile-directory dir 0 t)
      (when (and (fboundp 'native-comp-available-p) (native-comp-available-p))

    "Взять PATH и др. из оболочки (из кеша, если он свежий)."
    (if (and (file-exists-p my/shell-env-cache)
             (< (float-time (time-since (file-attribute-modification-time
                                         (file-attributes my/shell-env-cache))))
                my/shell-env-max-age))
        (progn
          (dolist (kv (with-temp-buffer
                        (insert-file-contents my/shell-env-cache)
                        (read (current-buffer))))
            (setenv (car kv) (cdr kv)))
          (setq exec-path (delete-dups
                           (append (split-string (or (getenv "PATH") "") path-separator t)
                                   exec-path))))
      (require 'exec-path-from-shell)
      (exec-path-from-shell-initialize)
      (with-temp-file my/shell-env-cache
        (prin1 (mapcar (lambda (v) (cons v (getenv v)))
                       exec-path-from-shell-variables)
               (current-buffer)))))

  (defun my/dashboard-insert-roam (list-size)
    "Секция: LIST-SIZE недавно изменённых заметок roam (n1, n2, …)."
    (dashboard-insert-heading "Недавние заметки (n1…):")
    (insert "\n")
    (setq my/dashboard--roam-files nil)
    (let ((dir (expand-file-name my/dashboard-roam-dir)))
      (if (not (file-directory-p dir))
          (insert "    (каталог заметок не найден)")
        (let* ((stamped (mapcar (lambda (f)
                                  (cons (file-attribute-modification-time (file-attributes f)) f))
                                (directory-files-recursively dir "\\.org\\'")))
               (sorted (seq-take (sort stamped (lambda (a b) (time-less-p (car b) (car a))))
                                 (min list-size 9)))
               (i 0))
          (dolist (entry sorted)
            (let ((f (cdr entry)))
              (setq i (1+ i))
              (push f my/dashboard--roam-files)
              (my/dashboard--insert-item "n" i f (my/dashboard--title f))))
          (setq my/dashboard--roam-files (nreverse my/dashboard--roam-files))))))

  ;; Отдельного пакета svg-mode нет, поэтому режим собран здесь: SVG — это XML, так что режим
  ;; наследует встроенный nxml-mode (подсветка, закрытие тегов, проверка разметки) и добавляет:
  ;;   - живой предпросмотр справа: обновляется через полсекунды после правки; если XML
  ;;     сломан, остаётся последняя удачная картинка, а в шапке окна — текст ошибки;
  ;;   - цвета вида #7dcfff подсвечиваются своим цветом (rainbow-mode);
  ;;   - C-c C-p — показать/убрать предпросмотр, C-c C-b — сменить фон предпросмотра
  ;;     (белый → серый → тёмный; на тёмном фоне чёрные иконки не видно),
  ;;   - C-c C-c — посмотреть файл как картинку (image-mode), там же C-c C-c вернёт в текст.
  ;; Нужна поддержка SVG в Emacs (librsvg): проверка — (image-type-available-p 'svg).
  (use-package rainbow-mode
    :commands rainbow-mode)

  (defvar svg-mode-big-size 300000
    "Файл больше стольких символов считается большим: без автопредпросмотра и подсветки цветов.")

  (defvar svg-mode-backgrounds '("#ffffff" "#c0c0c0" "#1a1b26")
    "Фоны предпросмотра, по кругу (C-c C-b).")
  (defvar svg-mode-wrap-attrs "auto"
    "Атрибуты внешнего <svg> для фрагмента без своего корня (C-c C-a).
\"auto\" — viewBox вычисляется по координатам x/y/x1/y1/cx/cy/r/width/height в коде;
пути (path d=…) при вычислении не учитываются — для них задайте viewBox вручную.")
  (defvar-local svg-mode-auto-wrap t
    "Не-nil: если в буфере нет корневого <svg>, предпросмотр оборачивает код в него.
Файл не меняется — оборачивается только то, что видно справа. Переключение: C-c C-w.")
  (defvar-local svg-mode--bg-index 0)
  (defvar-local svg-mode--preview nil "Буфер предпросмотра этого svg-буфера.")
  (defvar-local svg-mode--timer nil)

  (defun svg-mode--auto-attrs (text)
    "Атрибуты viewBox/width/height по рамке, в которую попадают координаты из TEXT."
    (let (xs ys lastx lasty)
      (with-temp-buffer
        (insert text)
        (goto-char (point-min))
        (while (re-search-forward
                "[ \t\n]\\(x1\\|x2\\|cx\\|x\\|y1\\|y2\\|cy\\|y\\|width\\|height\\|r\\)=[\"']\\(-?[0-9]*\\.?[0-9]+\\)"
                nil t)
          (let ((k (match-string 1)) (v (string-to-number (match-string 2))))
            (pcase k
              ((or "x" "x1" "x2" "cx") (setq lastx v) (push v xs))
              ((or "y" "y1" "y2" "cy") (setq lasty v) (push v ys))
              ("width"  (when lastx (push (+ lastx v) xs)))
              ("height" (when lasty (push (+ lasty v) ys)))
              ("r" (when (and lastx lasty)
                     (push (- lastx v) xs) (push (+ lastx v) xs)
                     (push (- lasty v) ys) (push (+ lasty v) ys)))))))
      (if (or (null xs) (null ys))
          "viewBox=\"0 0 100 100\" width=\"100\" height=\"100\""
        (let* ((pad 2)
               (x0 (- (apply #'min xs) pad)) (y0 (- (apply #'min ys) pad))
               (w (max 1 (+ (- (apply #'max xs) (apply #'min xs)) (* 2 pad))))
               (h (max 1 (+ (- (apply #'max ys) (apply #'min ys)) (* 2 pad)))))
          (format "viewBox=\"%s %s %s %s\" width=\"%s\" height=\"%s\""
                  x0 y0 w h w h)))))

  (defun svg-mode--payload (text)
    "TEXT для отрисовки. Убирает \\\" (скопировано из JSON-строки); если включено
автообрамление и корня <svg> нет — оборачивает."
    (let ((text (replace-regexp-in-string "\\\\\"" "\"" text t t)))
      (if (and svg-mode-auto-wrap
               (not (string-match-p "<svg[[:space:]>/]" text)))
          (concat "<svg xmlns=\"http://www.w3.org/2000/svg\" "
                  "xmlns:xlink=\"http://www.w3.org/1999/xlink\" "
                  (if (equal svg-mode-wrap-attrs "auto")
                      (svg-mode--auto-attrs text)
                    svg-mode-wrap-attrs)
                  ">\n" text "\n</svg>")
        text)))

  (defun svg-mode-unescape ()
    "Заменить в буфере \\\" на \" (текст, скопированный из JSON-строки)."
    (interactive)
    (save-excursion
      (goto-char (point-min))
      (let ((n 0))
        (while (search-forward "\\\"" nil t) (replace-match "\"" t t) (setq n (1+ n)))
        (message "Исправлено вхождений: %d" n))))

  ;; Масштаб и сдвиг предпросмотра. zoom = 1.0 — «вписать в окно»; сдвиг ox/oy — какая часть
  ;; увеличенной картинки видна (показывается срез картинки размером с окно).
  (defvar-local svg-mode--zoom 1.0)
  (defvar-local svg-mode--ox 0)
  (defvar-local svg-mode--oy 0)
  (defvar-local svg-mode--fit 1.0 "Масштаб «вписать в окно» с прошлой отрисовки.")

  (defvar-local svg-mode--error nil "Текст ошибки разбора SVG или nil.")

  (defun svg-mode--update (buf)
    "Перерисовать предпросмотр для svg-буфера BUF."
    (when (buffer-live-p buf)
      (with-current-buffer buf
        (let ((pb svg-mode--preview))
          (when (and (buffer-live-p pb) (get-buffer-window pb))
            (let* ((win  (get-buffer-window pb))
                   (ww   (max 50 (- (window-body-width win t) 4)))
                   (wh   (max 50 (- (window-body-height win t) 4)))
                   (data (svg-mode--payload
                          (buffer-substring-no-properties (point-min) (point-max))))
                   (bg   (nth svg-mode--bg-index svg-mode-backgrounds)))
              (condition-case err
                  (let* ((img0 (create-image data 'svg t :background bg))
                         (size (image-size img0 t)) ; здесь разбор падает, если SVG некорректен
                         (fit  (min (/ (float ww) (max 1 (car size)))
                                    (/ (float wh) (max 1 (cdr size)))))
                         (scale (* fit svg-mode--zoom))
                         (iw (round (* (car size) scale)))
                         (ih (round (* (cdr size) scale)))
                         (img (create-image data 'svg t :background bg :scale scale)))
                    (setq svg-mode--fit fit
                          svg-mode--error nil)
                    ;; сдвиг не должен выводить за край картинки
                    (setq svg-mode--ox (max 0 (min svg-mode--ox (max 0 (- iw ww))))
                          svg-mode--oy (max 0 (min svg-mode--oy (max 0 (- ih wh)))))
                    (with-current-buffer pb
                      (let ((inhibit-read-only t))
                        (erase-buffer)
                        (if (or (> iw ww) (> ih wh))
                            (insert-image img nil nil
                                          (list (buffer-local-value 'svg-mode--ox buf)
                                                (buffer-local-value 'svg-mode--oy buf)
                                                (min iw ww) (min ih wh)))
                          (insert-image img)))
                      (set-window-hscroll win 0)
                      (force-mode-line-update)))
                (error
                 (setq svg-mode--error (error-message-string err))
                 (with-current-buffer pb (force-mode-line-update))))))))))

  (defun svg-mode--pan (buf dx dy)
    "Сдвинуть видимую часть предпросмотра BUF на DX, DY пикселей."
    (with-current-buffer buf
      (setq svg-mode--ox (+ svg-mode--ox dx)
            svg-mode--oy (+ svg-mode--oy dy))
      (svg-mode--update buf)))

  (defun svg-mode--zoom-by (buf factor)
    "Умножить масштаб BUF на FACTOR, сохраняя в центре окна ту же точку картинки."
    (with-current-buffer buf
      (when-let ((win (and (buffer-live-p svg-mode--preview)
                           (get-buffer-window svg-mode--preview))))
        (let* ((ww (window-body-width win t)) (wh (window-body-height win t))
               (old (* svg-mode--fit svg-mode--zoom))
               (cx (/ (+ svg-mode--ox (/ ww 2.0)) old))
               (cy (/ (+ svg-mode--oy (/ wh 2.0)) old))
               (z (max 0.1 (min 50.0 (* svg-mode--zoom factor))))
               (new (* svg-mode--fit z)))
          (setq svg-mode--zoom z
                svg-mode--ox (round (- (* cx new) (/ ww 2.0)))
                svg-mode--oy (round (- (* cy new) (/ wh 2.0)))))
        (svg-mode--update buf))))

  (defun svg-mode--zoom-reset (buf)
    (with-current-buffer buf
      (setq svg-mode--zoom 1.0 svg-mode--ox 0 svg-mode--oy 0)
      (svg-mode--update buf)))

  (defun svg-mode-zoom-in ()    (interactive) (svg-mode--zoom-by (current-buffer) 1.25))
  (defun svg-mode-zoom-out ()   (interactive) (svg-mode--zoom-by (current-buffer) 0.8))
  (defun svg-mode-zoom-reset () (interactive) (svg-mode--zoom-reset (current-buffer)))

  ;; --- режим буфера предпросмотра: клавиши и мышь -------------------------------------
  (defvar-local svg-preview--source nil "svg-буфер, которому принадлежит этот предпросмотр.")

  (defmacro svg-preview--cmd (name doc &rest body)
    "Команда NAME в буфере предпросмотра; в BODY доступна переменная src."
    (declare (indent 2))
    `(defun ,name (&optional _e)
       ,doc (interactive)
       (when (buffer-live-p svg-preview--source)
         (let ((src svg-preview--source)) ,@body))))

  (svg-preview--cmd svg-preview-zoom-in  "Приблизить." (svg-mode--zoom-by src 1.25))
  (svg-preview--cmd svg-preview-zoom-out "Отдалить."   (svg-mode--zoom-by src 0.8))
  (svg-preview--cmd svg-preview-fit      "Вписать в окно." (svg-mode--zoom-reset src))
  (svg-preview--cmd svg-preview-left     "Сдвиг влево."  (svg-mode--pan src -60 0))
  (svg-preview--cmd svg-preview-right    "Сдвиг вправо." (svg-mode--pan src 60 0))
  (svg-preview--cmd svg-preview-up       "Сдвиг вверх."  (svg-mode--pan src 0 -60))
  (svg-preview--cmd svg-preview-down     "Сдвиг вниз."   (svg-mode--pan src 0 60))
  (svg-preview--cmd svg-preview-bg "Сменить фон."
    (with-current-buffer src (svg-mode-cycle-background)))
  (svg-preview--cmd svg-preview-refresh "Перерисовать." (svg-mode--update src))

  (defun svg-preview-drag (event)
    "Перетаскивание картинки мышью (зажать левую кнопку и двигать)."
    (interactive "e")
    (when (buffer-live-p svg-preview--source)
      (let ((src svg-preview--source)
            (last (posn-x-y (event-start event))) ev)
        (track-mouse
          (while (progn (setq ev (read-event)) (mouse-movement-p ev))
            (let ((cur (posn-x-y (event-start ev))))
              (when (and cur last (consp cur) (consp last))
                (svg-mode--pan src (- (car last) (car cur)) (- (cdr last) (cdr cur)))
                (setq last cur))))))))

  (defun svg-preview--header ()
    "Шапка окна предпросмотра: масштаб (или ошибка разбора SVG)."
    (let ((src svg-preview--source))
      (if (not (buffer-live-p src))
          ""
        (let ((err (buffer-local-value 'svg-mode--error src))
              (z   (buffer-local-value 'svg-mode--zoom src)))
          (if err
              (propertize (format " ошибка SVG: %s" err) 'face 'error)
            (format " %d%%   + − 0 — масштаб, колесо и перетаскивание — сдвиг" (round (* 100 z))))))))

  (defun svg-mode--schedule (&rest _)
    "После правки: перерисовать предпросмотр через 0.5 с простоя."
    (when (timerp svg-mode--timer) (cancel-timer svg-mode--timer))
    (when (buffer-live-p svg-mode--preview)
      (setq svg-mode--timer
            (run-with-idle-timer (if (< (buffer-size) svg-mode-big-size) 0.5 2.0)
                                 nil #'svg-mode--update (current-buffer)))))

  (defun svg-mode-preview ()
    "Показать предпросмотр справа или убрать его, если он уже виден."
    (interactive)
    (unless (image-type-available-p 'svg)
      (user-error "Эта сборка Emacs не умеет показывать SVG (нет librsvg)"))
    (let ((pb (if (buffer-live-p svg-mode--preview)
                  svg-mode--preview
                (setq svg-mode--preview
                      (let ((src (current-buffer)))
                        (with-current-buffer (generate-new-buffer
                                              (format "*svg: %s*" (buffer-name)))
                          (svg-preview-mode)
                          (setq svg-preview--source src)
                          (current-buffer)))))))
      (if-let ((win (get-buffer-window pb)))
          (delete-window win)
        (display-buffer-in-side-window
         pb '((side . right) (slot . 0) (window-width . 0.4)
              (window-parameters . ((no-delete-other-windows . t)))))
        (svg-mode--update (current-buffer)))))

  (defun svg-mode--auto-open (buf)
    "Открыть предпросмотр для BUF, когда буфер уже показан в окне (вызывается таймером)."
    (when (and (buffer-live-p buf) (get-buffer-window buf)
               (display-graphic-p) (image-type-available-p 'svg))
      (with-current-buffer buf
        (when (and (derived-mode-p 'svg-mode)
                   (not (buffer-live-p svg-mode--preview))
                   (< (buffer-size) svg-mode-big-size))
          (with-demoted-errors "svg-mode: предпросмотр: %S"
            (svg-mode-preview))))))

  (define-derived-mode svg-mode nxml-mode "SVG"
    "XML-режим для SVG с живым предпросмотром."
    ;; nxml по умолчанию проверяет SVG по схеме RELAX NG: на реальных файлах это очень
    ;; медленно и подвешивает Emacs. Проверку выключаем (flymake/предпросмотр и так покажут ошибку).
    (setq-local rng-nxml-auto-validate-flag nil)
    (add-hook 'after-change-functions #'svg-mode--schedule nil t)
    (svg-mode--update-name)
    (add-hook 'kill-buffer-hook
              (lambda ()
                (when (buffer-live-p svg-mode--preview)
                  (when-let ((w (get-buffer-window svg-mode--preview))) (delete-window w))
                  (kill-buffer svg-mode--preview)))
              nil t)
    ;; предпросмотр — не сейчас, а когда буфер уже показан в окне (иначе размер окна неизвестен,
    ;; а в демоне кадра может ещё не быть)
    (run-with-idle-timer 0.3 nil #'svg-mode--auto-open (current-buffer)))

  (add-hook 'svg-mode-hook
            (lambda ()
              ;; хуки родителя nxml-mode выполняются раньше этого — отключаем проверку после них
              (when (and (boundp 'rng-validate-mode) rng-validate-mode)
                (rng-validate-mode -1))
              (when (< (buffer-size) svg-mode-big-size)

  (defun morg--hashtagify ()
    "Показать группу тегов (группа 1 в данных совпадения) как чипы \" #a \"."
    (save-match-data
      (let ((beg (match-beginning 1))
            (end (match-end 1))
            colons)
        (save-excursion
          (goto-char beg)
          (while (search-forward ":" end t)
            (push (1- (point)) colons)))
        (setq colons (nreverse colons))
        (let ((n (length colons)) (i 0))
          (dolist (p colons)
            (let* ((prev (and (> i 0) (nth (1- i) colons)))
                   (next (and (< i (1- n)) (nth (1+ i) colons)))
                   (face-prev (and prev (morg--tag-face
                                         (buffer-substring-no-properties (1+ prev) p))))
                   (face-next (and next (morg--tag-face
                                         (buffer-substring-no-properties (1+ p) next)))))
              ;; двоеточие = закрыть предыдущий чип + зазор + открыть следующий
              (put-text-property
               p (1+ p) 'display
               (concat (and face-prev (propertize " " 'face face-prev))
                       (and face-prev face-next " ")
                       (and face-next (propertize " #" 'face face-next))))
              (when face-next
                (add-face-text-property (1+ p) next face-next)))
            (setq i (1+ i))))))
    nil)

  (defface morg-table-header-row

  (defun morg--table-header-matcher (limit)
    "Найти строку-шапку таблицы до LIMIT (для font-lock)."
    (let (found)
      (while (and (not found)
                  (re-search-forward "^[ \t]*|[^-\n].*$" limit t))
        (let* ((md   (match-data))
               (beg  (match-beginning 0))
               (prev-ok (save-excursion
                          (goto-char beg)
                          (or (bobp)
                              (progn (forward-line -1)
                                     (not (looking-at-p "[ \t]*|"))))))
               (next-ok (save-excursion
                          (goto-char beg)
                          (forward-line 1)
                          (looking-at-p "[ \t]*|-"))))
          (when (and prev-ok next-ok)
            (set-match-data md)
            (setq found t))))
      found))

  ;;   |---+---|             -> ├───┼───┤  (┌┬┐ если это верх таблицы, └┴┘ — низ)
  ;; Одно и то же начертание у вертикальных и горизонтальных линий, поэтому
  ;; они одной толщины. У строк таблицы отключается межстрочный интервал
  ;; (line-height t на переводе строки), чтобы линии не рвались между строками.
  (defvar morg-table-line-color "#7d88b8"
    "Цвет линий таблиц (nil — цвет текста).")

  (defun morg--table-glyph (ch pos beg last top bottom)
    "Символ-рамка для CH в строке-линии: POS — позиция, BEG/LAST — края строки."
    (cond ((eq ch ?-) "─")
          ((eq ch ?+) (cond (top "┬") (bottom "┴") (t "┼")))
          ((= pos beg)  (cond (top "┌") (bottom "└") (t "├")))
          ((= pos last) (cond (top "┐") (bottom "┘") (t "┤")))
          (t "│")))

  (defun morg--table-prettify (limit)
    "Для font-lock: нарисовать линии таблиц Org в области до LIMIT."
    (while (re-search-forward "^[ \t]*\\(|.*\\)$" limit t)
      (let* ((beg  (match-beginning 1))
             (bol  (line-beginning-position))
             (end  (save-excursion
                     (goto-char (match-end 1))
                     (skip-chars-backward " \t")
                     (point)))
             (last (1- end))
             (hline (save-excursion (goto-char beg) (looking-at-p "|-[-+|]*[ \t]*$")))
             (top    (save-excursion (goto-char bol)
                                     (or (bobp)
                                         (progn (forward-line -1)
                                                (not (looking-at-p "[ \t]*|"))))))
             (bottom (save-excursion (goto-char bol)
                                     (forward-line 1)
                                     (not (looking-at-p "[ \t]*|")))))
        (save-excursion
          (goto-char beg)
          (while (< (point) end)
            (let ((ch (char-after)))
              (when (or (eq ch ?|) (and hline (memq ch '(?- ?+))))
                (put-text-property
                 (point) (1+ (point)) 'display
                 (propertize (if hline
                                 (morg--table-glyph ch (point) beg last top bottom)
                               "│")
                             'face (and morg-table-line-color
                                        `(:foreground ,morg-table-line-color))))))
            (forward-char 1)))
        ;; line-height = t на символе перевода строки: у этой строки межстрочный
        ;; интервал игнорируется (свойство line-spacing лишь увеличивает интервал,
        ;; уменьшить им нельзя), поэтому вертикальные линии не рвутся
        (let ((eol (line-end-position)))
          (when (< eol (point-max))
            (put-text-property eol (1+ eol) 'line-height t)))))
    nil)

  (with-eval-after-load 'org
    (font-lock-add-keywords 'org-mode '(morg--table-prettify) t))

  ;; --- callout-блоки как в Obsidian ------------------------------------------
  ;; #+begin_note ... #+end_note (а также tip / warning / important / question):
  ;; тело блока слегка подкрашено цветом типа, строки begin/end — цветные и жирные.
  ;; Внутрь можно класть обычные src-блоки (см. шаблоны <nsql, <nsh, <nel).
  (defvar morg-callout-colors
    '(("note"      . "#7dcfff")
      ("tip"       . "#9ece6a")
      ("warning"   . "#e0af68")
      ("important" . "#f7768e")
      ("question"  . "#7aa2f7")
      ("quote"     . "#9aa5ce")            ; цитаты: курсив, приглушённый серо-синий
      ("example"   . "#73daca"))
    "Типы callout-блоков и их цвета.")
  (defvar morg-callout-body-tint 0.08 "Насколько цвет типа подмешан в фон тела блока.")

  (defvar morg--callout-names
    (regexp-opt (mapcar #'car morg-callout-colors))
    "Регулярное выражение с именами callout-блоков.")

  (defun morg--callout-color (name)
    "Цвет типа блока NAME."
    (or (cdr (assoc (downcase name) morg-callout-colors)) "#7fd6e8"))

  (defun morg--callout-body-face ()
    "Лицо тела блока; тип берётся из группы 1 совпадения morg--callout-matcher."
    `(:background ,(morg--mix (morg--callout-color (match-string-no-properties 1))
                              (morg--base-bg)
                              morg-callout-body-tint)
      ,@(when (string-equal-ignore-case (match-string-no-properties 1) "quote")
          '(:slant italic))                 ; цитаты курсивом
      :extend t))

  (defun morg--callout-head-face (name)
    "Лицо строк #+begin_/#+end_ для типа NAME."
    ;; Только цвет и жирность, без своего фона: фон строк даёт тело блока,
    ;; иначе он накладывается на фон иконки org-modern (двойная подсветка).
    `(:foreground ,(morg--callout-color name) :weight bold))

  (defun morg--callout-matcher (limit)
    "Для font-lock: найти блок #+begin_X ... #+end_X; совпадение 0 — весь блок, 1 — тип."
    (let (found)
      (while (and (not found)
                  (re-search-forward
                   (concat "^[ \t]*#\\+begin_\\(" morg--callout-names "\\)\\_>") limit t))
        (let ((bb (match-beginning 0))
              (tb (match-beginning 1))
              (te (match-end 1))
              (name (match-string-no-properties 1)))
          (when (re-search-forward
                 (concat "^[ \t]*#\\+end_" (regexp-quote name) "\\_>.*$") nil t)
            ;; тело включает строку #+end_ вместе с переводом строки: иначе
            ;; :extend не срабатывает и красится только видимый символ-замена
            (set-match-data (list bb (min (point-max) (1+ (point))) tb te))
            (setq found t))))
      found))

  (defun morg--callout-extend-region ()
    "Если правка внутри callout-блока, перекрасить блок целиком."
    (let ((re (concat "^[ \t]*#\\+\\(begin\\|end\\)_" morg--callout-names "\\_>"))
          (changed nil))
      (save-excursion
        (goto-char font-lock-beg)
        (when (and (re-search-backward re nil t)
                   (string-equal-ignore-case (match-string 1) "begin"))
          (setq font-lock-beg (line-beginning-position) changed t)))
      (save-excursion
        (goto-char font-lock-end)
        (when (and (re-search-forward re nil t)
                   (string-equal-ignore-case (match-string 1) "end"))
          (setq font-lock-end (line-end-position) changed t)))
      changed))

  (defun morg--callout-src-fontify (limit)
    "Подсветить синтаксис src-блоков, вложенных в callout (Org сам их не трогает).
Только красит и всегда возвращает nil — для font-lock это «ничего не найдено»."
    (while (re-search-forward
            "^[ \t]*#\\+begin_src[ \t]+\\([^ \t\n]+\\)[^\n]*\n" limit t)
      (let ((lang (match-string-no-properties 1))
            (beg (match-end 0)))
        (when (and (save-excursion
                     (goto-char beg)
                     (and (re-search-backward
                           (concat "^[ \t]*#\\+\\(begin\\|end\\)_" morg--callout-names "\\_>")
                           nil t)
                          (string-equal-ignore-case (match-string 1) "begin")))
                   (re-search-forward "^[ \t]*#\\+end_src" nil t))
          (let ((end (match-beginning 0)))
            (ignore-errors (org-src-font-lock-fontify-block lang beg end))))))
    nil)

  (font-lock-add-keywords
   'org-mode
   `(morg--callout-src-fontify          ; раньше фона: фон блока накладывается поверх
     (morg--callout-matcher (0 (morg--callout-body-face) prepend))
     (,(concat "^[ \t]*\\(#\\+\\(?:begin\\|end\\)_\\(" morg--callout-names "\\)\\_>.*\\)$")
      (1 (morg--callout-head-face (match-string-no-properties 2)) prepend)))
   t)

  (add-hook 'org-mode-hook
            (lambda ()
              (add-hook 'font-lock-extend-region-functions
                        #'morg--callout-extend-region t t)))

  (defvar morg-callout-src-indent 2 "Отступ вложенного в callout src-блока.")

  (defun morg--in-callout-p ()
    "Не-nil, если точка внутри callout-блока (между #+begin_X и #+end_X)."
    (save-excursion
      (beginning-of-line)
      (and (re-search-backward
            (concat "^[ \t]*#\\+\\(begin\\|end\\)_" morg--callout-names "\\_>") nil t)
           (string-equal-ignore-case (match-string 1) "begin"))))

  (defun morg--tempo-indent-nested (fn &rest args)
    "После <s TAB (и др.) внутри callout сдвинуть вставленный src-блок."
    (let ((inside (morg--in-callout-p))
          (start (line-beginning-position))
          (marker (point-marker)))
      (apply fn args)
      (when (and inside (> (point-max) start))
        (save-excursion
          (goto-char start)
          (let* ((ind (max morg-callout-src-indent (current-indentation)))
                 (end (when (re-search-forward "^[ \t]*#\\+end_src" nil t)
                        (copy-marker (line-end-position)))))
            (when end
              (goto-char start)
              (while (<= (point) end)
                (unless (or (looking-at-p "[ \t]*$") (>= (current-indentation) ind))
                  (indent-line-to ind))
                (forward-line 1)))))
        (when (looking-at-p "[ \t]*$")      ; точка на пустой строке кода — встать после отступа
          (end-of-line)))
      (set-marker marker nil)))

  (with-eval-after-load 'org-tempo
    (advice-add 'org-tempo-complete-tag :around #'morg--tempo-indent-nested))

  (defun morg-indent-nested-src ()
    "Сдвинуть на `morg-callout-src-indent' все src-блоки внутри callout в буфере."
    (interactive)
    (save-excursion
      (goto-char (point-min))
      (let ((n 0))
        (while (re-search-forward "^\\([ \t]*\\)#\\+begin_src\\_>" nil t)
          (let ((bol (match-beginning 0))
                (cur (current-indentation)))
            (when (and (< cur morg-callout-src-indent)
                       (save-excursion (goto-char bol) (morg--in-callout-p)))
              (let ((shift (- morg-callout-src-indent cur))
                    (end (save-excursion
                           (when (re-search-forward "^[ \t]*#\\+end_src" nil t)
                             (copy-marker (line-end-position))))))
                (when end
                  (goto-char bol)
                  (while (<= (point) end)
                    (unless (looking-at-p "[ \t]*$")
                      (insert (make-string shift ?\s)))
                    (forward-line 1))
                  (setq n (1+ n)))))))
        (message "Сдвинуто src-блоков в callout: %d" n))))
  ;; C-c n i обычно занят org-roam-node-insert, поэтому C-c n >
  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-c n >") #'morg-indent-nested-src))

  ;; --- #+caption: как заголовок блока ----------------------------------------------
  ;; Подпись над src-блоком / таблицей: яркий жирный текст на мягком фоне во всю
  ;; ширину, сразу над блоком. Описание «что делает блок» пишите так:
  ;;   #+caption: Создать и поднять мост
  ;;   #+begin_src shell ...
  (defvar morg-caption-color "#7dcfff" "Цвет подписи #+caption:.")
  (defvar morg-caption-tint  0.10     "Насколько цвет подписи подмешан в фон строки.")

  (defun morg--caption-line-face ()
    `(:background ,(morg--mix morg-caption-color (morg--base-bg) morg-caption-tint)

  ;; --- src-блоки: служебные строки скрыты, пока курсор не в блоке ---------------
  ;; Вне блока он выглядит так:
  ;;     ▍ shell  Создать и поднять мост      ← язык (приглушённо) и текст #+caption:
  ;;     ip l a br0 type bridge ...           ← код
  ;; Строки #+begin_src и #+end_src и параметры (:results, :dir …) не видны.
  ;; Если подписи нет, остаётся строка «<> язык» без параметров.
  ;; Как только курсор оказывается на подписи, на begin/end или в коде, всё показывается.
  (defvar morg-hide-src-delimiters t
    "Прятать служебные строки src-блоков, пока курсор не внутри блока.")
  (defvar morg-caption-lang-color "#565f89" "Цвет названия языка в подписи.")

  (defconst morg--affil-re
    "^[ \t]*#\\+\\(?:caption\\|name\\|header\\|attr_[[:alnum:]_]+\\):"
    "Строки-описания, которые стоят над блоком (подпись, имя, заголовки).")

  (defvar-local morg--active-block nil
    "(НАЧАЛО . КОНЕЦ) src-блока под курсором вместе с подписью, иначе nil.")

  (defun morg--src-bounds ()
    "(НАЧАЛО . КОНЕЦ) src-блока под точкой вместе со строками-подписями над ним, иначе nil."
    (save-excursion
      (beginning-of-line)
      (let (b e)
        (while (looking-at-p morg--affil-re) (forward-line 1))
        (cond
         ((looking-at-p "[ \t]*#\\+begin_src\\_>")
          (setq b (point)))
         ((looking-at-p "[ \t]*#\\+end_src\\_>")
          (setq e (line-end-position))
          (when (and (re-search-backward "^[ \t]*#\\+\\(begin\\|end\\)_src\\_>" nil t)
                     (string-equal-ignore-case (match-string 1) "begin"))
            (setq b (match-beginning 0))))
         (t
          (end-of-line)
          (when (and (re-search-backward "^[ \t]*#\\+\\(begin\\|end\\)_src\\_>" nil t)
                     (string-equal-ignore-case (match-string 1) "begin"))
            (setq b (match-beginning 0)))))
        (when b
          (unless e
            (goto-char b)
            (when (re-search-forward "^[ \t]*#\\+end_src\\_>.*$" nil t)
              (setq e (point))))
          (when e
            (goto-char b)
            (while (and (not (bobp))
                        (progn (forward-line -1) (looking-at-p morg--affil-re)))
              (setq b (point)))
            (cons b e))))))

  (defun morg--block-active-p (pos)
    "Не-nil, если POS внутри блока, на котором сейчас курсор."
    (and morg--active-block
         (<= (car morg--active-block) pos (cdr morg--active-block))))

  (defun morg--update-active-block ()
    "После каждой команды: запомнить блок под курсором и перерисовать затронутые блоки."
    (when morg-hide-src-delimiters
      (let ((new (morg--src-bounds))
            (old morg--active-block))
        (unless (equal new old)
          (setq morg--active-block new)
          (dolist (r (list old new))
            (when r
              (font-lock-flush (max (point-min) (car r))
                               (min (point-max) (1+ (cdr r))))))))))

  (defun morg--caption-above-p (pos)
    "Не-nil, если над строкой в POS (через строки-описания) есть #+caption: или #+name:."
    (save-excursion
      (goto-char pos)
      (let (found (case-fold-search t))
        (while (and (not (bobp))
                    (progn (forward-line -1) (looking-at-p morg--affil-re)))
          (when (looking-at-p "[ \t]*#\\+\\(?:caption\\|name\\):") (setq found t)))
        found)))

  (defun morg--hide-line (bol eol)
    "Убрать строку BOL…EOL вместе с переводом строки (display \"\")."
    (put-text-property bol (min (point-max) (1+ eol)) 'display ""))

  ;; --- #+name: у src-блока ------------------------------------------------------------
  ;; Имя блока (#+name: fetch) нужно, чтобы ссылаться на него из других блоков
  ;; (:var x=fetch, <<fetch>>, #+call:). Вне блока оно не выводится отдельной строкой:
  ;;   - есть #+caption: — имя показывается в подписи:  ▍ shell  ‹fetch›  Текст подписи
  ;;   - подписи нет    — само имя становится подписью: ▍ shell  fetch
  ;; Под курсором (внутри блока) показываются настоящие строки.
  (defvar morg-name-color "#bb9af7" "Цвет имени блока (#+name:) в подписи.")

  (defun morg--affil-group (pos)
    "Для группы строк-описаний, в которой лежит строка POS: (ИМЯ CAPTION-P ЯЗЫК).
ЯЗЫК — язык src-блока, который идёт сразу за группой, иначе nil."
    (save-match-data
      (save-excursion
        (let ((case-fold-search t) name cap)
          (goto-char pos)
          (beginning-of-line)
          ;; подняться к началу группы
          (while (and (not (bobp))
                      (save-excursion (forward-line -1) (looking-at-p morg--affil-re)))
            (forward-line -1))
          ;; пройти группу вниз
          (while (looking-at-p morg--affil-re)
            (cond ((looking-at "[ \t]*#\\+name:[ \t]*\\(.*?\\)[ \t]*$")
                   (setq name (match-string-no-properties 1)))
                  ((looking-at-p "[ \t]*#\\+caption:") (setq cap t)))
            (forward-line 1))
          (list (and name (not (string-empty-p name)) name)
                cap
                (when (looking-at "[ \t]*#\\+begin_src[ \t]+\\([^ \t\n]+\\)")
                  (match-string-no-properties 1)))))))

  (defvar morg-vars-max-width 70 "Максимальная ширина показа переменных :var в подписи.")
  (defvar morg-vars-color "#e0af68" "Цвет строки с переменными :var под подписью блока.")

  (defun morg--vars-marker ()
    "Иконка «tune» перед переменными; если глиф не отображается — «ƒ»."
    (let ((icon (and (display-graphic-p) (fboundp 'nerd-icons-mdicon)
                     (ignore-errors (nerd-icons-mdicon "nf-md-tune")))))
      (if (and (stringp icon) (> (length icon) 0) (char-displayable-p (aref icon 0)))
          ;; шрифт иконки (face от nerd-icons) сохраняем, цвет — наш
          (propertize icon 'face (list `(:foreground ,morg-vars-color)
                                       (get-text-property 0 'face icon)))
        (propertize "ƒ" 'face `(:foreground ,morg-vars-color)))))

  (defun morg--block-vars-string (begin-bol)
    "Строка «  <иконка> x=fetch  ·  n=5» из :var заголовка src-блока, чья строка #+begin_src начинается в
BEGIN-BOL; nil, если переменных нет."
    (save-excursion
      (goto-char begin-bol)
      (when (looking-at "[ \t]*#\\+begin_src[ \t]+[^ \t\n]+\\(.*\\)$")
        (let (vars in)
          (dolist (tok (split-string (match-string-no-properties 1)))
            (cond ((string-prefix-p ":" tok) (setq in (equal tok ":var")))
                  (in (push tok vars))))
          (when vars
            (concat
             "  " (morg--vars-marker) " "
             (propertize
              (truncate-string-to-width
               (string-join (nreverse vars) "  ·  ") morg-vars-max-width nil nil "…")
              'face `(:foreground ,morg-vars-color :weight normal :slant normal))))))))

  (defun morg--caption-block-begin (pos)

      (while (looking-at-p "[ \t]*#\\+[nN][aA][mM][eE]:") (forward-line 1))

  (defun morg--style-caption ()
    "font-lock: #+caption:. Группа 1 — «#+caption: » с пробелами; показать «▍ язык  ‹имя›  »."
    (let ((bol (match-beginning 0)) (eol (match-end 0))
          (pb (match-beginning 1)) (pe (match-end 1)))
      (when (and morg-hide-src-delimiters (not (morg--block-active-p bol)))
        (let* ((grp  (morg--affil-group bol))
               (name (nth 0 grp))
               (lang (nth 2 grp)))
          (put-text-property
           pb pe 'display
           (concat (propertize "▍ " 'face `(:foreground ,morg-caption-color :weight bold))
                   (when lang
                     (propertize (concat lang "  ")
                                 'face `(:foreground ,morg-caption-lang-color
                                         :weight normal :slant normal)))
                   (when (and name lang)
                     (propertize (concat "‹" name "›  ")
                                 'face `(:foreground ,morg-name-color
                                         :weight normal :slant normal)))))
        ;; переменные :var блока — после подписи (свёрнутый блок: см. morg--style-src-begin)
        (let ((b (morg--caption-block-begin bol)))
          (when (and b (< eol (point-max))
                     (not (and (fboundp 'org-fold-folded-p)
                               (org-fold-folded-p (save-excursion (goto-char b)
                                                                  (line-end-position))))))
            (let ((v (morg--block-vars-string b)))
              (when v (put-text-property eol (1+ eol) 'display (concat "\n" v "\n")))))))))
    nil)

  (defun morg--style-name ()
    "font-lock: #+name:. Группа 1 — «#+name: ». Только для имён src-блоков."
    (let ((bol (match-beginning 0)) (eol (match-end 0))
          (pb (match-beginning 1)) (pe (match-end 1)))
      (when (and morg-hide-src-delimiters (not (morg--block-active-p bol)))
        (let* ((grp  (morg--affil-group bol))
               (cap  (nth 1 grp))
               (lang (nth 2 grp)))
          (when lang                      ; имя таблицы и т. п. не трогаем
            (if cap
                (morg--hide-line bol eol) ; имя уйдёт в подпись
              ;; подписи нет — имя играет её роль
              (put-text-property
               pb pe 'display
               (concat (propertize "▍ " 'face `(:foreground ,morg-caption-color :weight bold))
                       (propertize (concat lang "  ")
                                   'face `(:foreground ,morg-caption-lang-color
                                           :weight normal :slant normal))))
              (add-face-text-property bol (min (point-max) (1+ eol))
                                      (morg--caption-line-face) t)
              (add-face-text-property pe eol
                                      `(:foreground ,morg-name-color :weight bold) t))))))
    nil)

  (defun morg--style-src-begin ()
    "font-lock: #+begin_src. Группа 1 — параметры после языка."
    (let ((bol (match-beginning 0)) (eol (match-end 0))
          (pb (match-beginning 1)) (pe (match-end 1)))
      (when (and morg-hide-src-delimiters (not (morg--block-active-p bol)))
        (if (morg--caption-above-p bol)
            (if (and (fboundp 'org-fold-folded-p) (org-fold-folded-p eol))
                ;; блок свёрнут: прячем и перевод строки подписи, чтобы стрелка свёртки
                ;; (org-ellipsis) оказалась на строке подписи, сразу после текста
                (let ((pos bol) (v (morg--block-vars-string bol)))
                  (if v
                      ;; есть :var — строка begin_src превращается в строку переменных,
                      ;; стрелка свёртки встаёт в её конце
                      (put-text-property bol eol 'display v)
                    (put-text-property bol eol 'display "")
                    (save-excursion
                      (goto-char bol)
                      (while (and (> pos (point-min))
                                  (progn (forward-line -1)
                                         (looking-at-p "[ \t]*#\\+[nN][aA][mM][eE]:")))
                        (setq pos (point))))
                    (when (> pos (point-min))
                      (put-text-property (1- pos) pos 'display ""))))
              (morg--hide-line bol eol))               ; язык уже в подписи
          (when (< pb pe) (put-text-property pb pe 'invisible t)))))
    nil)

  (defun morg--style-src-end ()

    nil)

  ;; S-TAB (org-global-cycle) ходит по кругу: обзор → оглавление → показать всё.

  ;; --- таблицы: подсветка текущей строки и столбца ---------------------------------
  ;; Пока курсор в таблице, строка и столбец с курсором слегка подкрашены
  ;; (на пересечении — ярче). Выключить: (setq morg-table-highlight nil).
  (defvar morg-table-highlight t)
  (defvar morg-table-hl-color "#7aa2f7")
  (defvar-local morg--table-overlays nil)
  (defvar-local morg--table-hl-state nil)

  (defun morg--table-hl-clear ()
    (mapc #'delete-overlay morg--table-overlays)
    (setq morg--table-overlays nil
          morg--table-hl-state nil))

  (defun morg--table-hl-update ()
    "После каждой команды: подсветить строку и столбец таблицы под курсором."
    (when morg-table-highlight
      (if (not (org-at-table-p))
          (when morg--table-overlays (morg--table-hl-clear))
        (let* ((bol (line-beginning-position))
               (cell (max 1 (how-many "|" bol (point))))
               (state (list bol cell (buffer-chars-modified-tick))))
          (unless (equal state morg--table-hl-state)
            (morg--table-hl-clear)
            (setq morg--table-hl-state state)
            (let ((row-face `(:background ,(morg--mix morg-table-hl-color (morg--base-bg) 0.14)
                              :extend t))
                  (col-face `(:background ,(morg--mix morg-table-hl-color (morg--base-bg) 0.10)))
                  ov)
              ;; текущая строка
              (setq ov (make-overlay bol (min (point-max) (1+ (line-end-position)))))
              (overlay-put ov 'face row-face)
              (overlay-put ov 'priority 5)
              (push ov morg--table-overlays)
              ;; столбец во всех строках таблицы (кроме разделителей |---+---|)
              (save-excursion
                (goto-char (org-table-begin))
                (let ((end (org-table-end)))
                  (while (< (point) end)
                    (unless (org-at-table-hline-p)
                      (let ((eol (line-end-position)) pipes)
                        (save-excursion
                          (while (search-forward "|" eol t)
                            (push (1- (point)) pipes)))
                        (setq pipes (nreverse pipes))
                        (let ((s (nth (1- cell) pipes))
                              (e (nth cell pipes)))
                          (when (and s e)
                            (setq ov (make-overlay (1+ s) e))
                            (overlay-put ov 'face col-face)
                            (overlay-put ov 'priority 6)
                            (push ov morg--table-overlays)))))
                    (forward-line 1))))))))))

  (defun morg--setup-table-hl ()

  (add-hook 'org-mode-hook #'morg--setup-table-hl)

  ;; --- определения: «Термин — описание» ----------------------------------------------
  ;; Автоматически выделяется термин (1–4 слова) перед тире « — » или « – » (с пробелами)
  ;; в начале строки, пункта списка или предложения. Слово «это» писать не нужно.
  ;; Дефис « - » намеренно не считается: он слишком часто встречается в обычном тексте.
  ;; Не срабатывает в заголовках, таблицах и src-блоках.
  ;; Выключить: (setq morg-definition-highlight nil), затем M-x font-lock-flush.
  (defvar morg-definition-highlight t)

  (defface morg-definition-term
    '((t :foreground "#ff9e64" :weight bold))
    "Лицо термина в определении «Термин — описание».")

  (defconst morg--definition-re
    (concat
     "\\(?:^[ \t]*\\(?:\\(?:[-+*]\\|[0-9]+[.)]\\)[ \t]+\\)?"   ; начало строки / пункта списка
     "\\|[.!?][ \t]+\\)"                                        ; или начало предложения
     "\\(\\(?:[[:alnum:]_/]+\\(?:[-.][[:alnum:]_/]+\\)*\\)"     ; первое слово (можно с - и .)
     "\\(?:[ \t]+[[:alnum:]_/]+\\(?:[-.][[:alnum:]_/]+\\)*\\)\\{0,3\\}"  ; ещё до трёх слов
     "\\(?:[ \t]+([^()\n]\\{1,40\\})\\)?\\)"                           ; необязательно: (перевод)
     "\\(?:[ \t]+[~=][^~=\n]+[~=]\\)?"                                 ; необязательно: ~код~ (не в группе 1)
     "[ \t]+[—–]\\(?:[ \t]\\|$\\)")                                    ; тире, дальше пробел или конец строки
    "Регулярное выражение определения; группа 1 — термин.")

  (defun morg--definition-face ()
    "Font-lock: вернуть лицо термина (группа 1) или nil, если здесь подсвечивать не надо."
    (when morg-definition-highlight
      (save-match-data
        (save-excursion
          (goto-char (match-beginning 1))
          (unless (or (org-at-heading-p)
                      (org-at-table-p)
                      (org-in-src-block-p)
                      (save-excursion        ; строки #+caption:, #+begin_… и т. п.
                        (beginning-of-line)
                        (looking-at-p "[ \t]*#")))
            'morg-definition-term)))))

  (with-eval-after-load 'org
    (font-lock-add-keywords
     'org-mode `((,morg--definition-re (1 (morg--definition-face) prepend))) t))

  ;; --- язык src-блока -> режим для подсветки -----------------------------------------
  ;; Org по имени языка сам ищет режим вида <язык>-mode. Режима rust-mode у нас нет
  ;; (только встроенный rust-ts-mode), а смену major-mode-remap-alist Org не учитывает —
  ;; поэтому в #+begin_src rust подсветки не было. Указываем режимы явно:
  ;; запись ("rust" . rust-ts) означает режим rust-ts-mode (Org сам добавляет "-mode").
  (with-eval-after-load 'org
    (dolist (pair '(("rust" . rust-ts)
                    ("c"    . c-ts)
                    ("C"    . c-ts)
                    ("js"         . js-ts)
                    ("javascript" . js-ts)
                    ("ts"         . typescript-ts)
                    ("typescript" . typescript-ts)
                    ;; zsh/bash/shell подсвечиваются режимом sh-mode (отдельного zsh-mode нет)
                    ("zsh"   . sh)
                    ("bash"  . sh)
                    ("shell" . sh)))
      (setf (alist-get (car pair) org-src-lang-modes nil nil #'equal) (cdr pair)))
    ;; json: режима json-mode в Emacs нет, и блок #+begin_src json не подсвечивался.
    ;; Берём json-ts-mode (если установлена грамматика json), иначе встроенный js-json-mode.
    (setf (alist-get "json" org-src-lang-modes nil nil #'equal)
          (if (and (fboundp 'treesit-language-available-p)
                   (treesit-language-available-p 'json))
              'json-ts
            'js-json)))

  ;; --- свёртка объектов и массивов JSON прямо в блоке ----------------------------------
  ;; В src-блоке json (в том числе в результате, :wrap src json) TAB на строке, которая
  ;; заканчивается на «{» или «[», сворачивает всё до парной скобки в «{ … }»; ещё раз TAB —
  ;; раскрывает. C-c n j — свернуть/раскрыть все вложенные объекты блока сразу.
  ;; Свёртка — оверлеи (временные пометки поверх текста): файл не меняется.
  (defun morg--json-open-pos ()
    "Позиция «{»/«[» в конце текущей строки или nil."
    (save-excursion
      (end-of-line)
      (skip-chars-backward " \t")
      (when (memq (char-before) '(?{ ?\[))
        (1- (point)))))

  (defun morg--json-match (open limit)
    "Позиция скобки, парной OPEN (строки в кавычках пропускаются), не дальше LIMIT."
    (save-excursion
      (goto-char (1+ open))
      (let ((depth 1) found)
        (while (and (not found) (< (point) limit)
                    (re-search-forward "[]\"[{}]" limit t))
          (let ((c (char-before)))
            (cond ((eq c ?\")
                   (while (and (< (point) limit) (not (eq (char-after) ?\")))
                     (forward-char (if (eq (char-after) ?\\) 2 1)))
                   (forward-char 1))
                  ((memq c '(?{ ?\[)) (setq depth (1+ depth)))
                  (t (setq depth (1- depth))
                     (when (zerop depth) (setq found (1- (point))))))))
        found)))

  (defun morg--json-fold-at (open limit)
    "Свернуть (или раскрыть, если уже свёрнуто) контейнер с открывающей скобкой в OPEN."
    (let ((ov (cl-find-if (lambda (o) (and (overlay-get o 'morg-jfold)
                                           (= (overlay-start o) (1+ open))))
                          (overlays-at (1+ open)))))
      (if ov
          (progn (delete-overlay ov) 'unfolded)
        (when-let ((close (morg--json-match open limit)))
          (when (> close (1+ open))
            (let ((o (make-overlay (1+ open) close nil t nil)))
              (overlay-put o 'morg-jfold t)
              (overlay-put o 'evaporate t)
              (overlay-put o 'display (propertize " … " 'face 'shadow))
              (overlay-put o 'isearch-open-invisible #'delete-overlay)
              'folded))))))

  (defun morg--json-block ()
    "(НАЧАЛО . КОНЕЦ) содержимого json-блока под точкой или nil."
    (when (and (derived-mode-p 'org-mode) (org-in-src-block-p))
      (let ((el (org-element-at-point)))
        (when (and (eq (org-element-type el) 'src-block)
                   (equal (downcase (or (org-element-property :language el) "")) "json"))
          (cons (org-element-property :post-affiliated el)
                (org-element-property :end el))))))

  (defun morg-json-fold-toggle ()
    "TAB в json-блоке на строке с «{»/«[» в конце: свернуть/раскрыть. Иначе nil (обычный TAB)."
    (when-let ((blk (morg--json-block))
               (open (morg--json-open-pos)))
      (morg--json-fold-at open (cdr blk))
      t))

  (defun morg-json-fold-all ()
    "Свернуть все вложенные объекты json-блока под точкой; если уже что-то свёрнуто — раскрыть всё."
    (interactive)
    (let ((blk (or (morg--json-block) (user-error "Поставьте курсор в src-блок json")))
          (any nil))
      (dolist (o (overlays-in (car blk) (cdr blk)))
        (when (overlay-get o 'morg-jfold) (setq any t) (delete-overlay o)))
      (unless any
        (save-excursion
          (goto-char (car blk))
          (forward-line 2)                        ; верхний контейнер оставляем открытым
          (while (< (point) (cdr blk))
            (when-let ((open (morg--json-open-pos)))
              (morg--json-fold-at open (cdr blk)))
            (forward-line 1))))
      (message (if any "JSON раскрыт" "Вложенные объекты свёрнуты (TAB на строке — раскрыть)"))))

  (with-eval-after-load 'org
    (add-hook 'org-tab-first-hook #'morg-json-fold-toggle)
    (define-key org-mode-map (kbd "C-c n j") #'morg-json-fold-all))

  ;; --- результат JSON подсвечивается как JSON ----------------------------------------
  ;; Обычный :results output вставляется строками «: …» без подсветки. Если вывод блока —
  ;; корректный JSON, он вставляется как #+begin_src json … #+end_src, и Org подсвечивает
  ;; его режимом json. Определение «это JSON» — по настоящему разбору (json-parse-string),
  ;; а не по первому символу. Не применяется, если в :results уже указан формат
  ;; (verbatim, raw, table, code, silent …) — так можно отключить для одного блока:
  ;;   #+begin_src zsh :results output verbatim replace
  (defvar morg-results-auto-json t "Подсвечивать результат-JSON как src-блок json.")
  (defvar morg-results-json-max 300000 "Не разбирать вывод длиннее стольких символов.")

  (defun morg--json-string-p (s)
    (and (stringp s)
         (< (length s) morg-results-json-max)
         (string-match-p "\\`[ \t\n]*[{[]" s)
         (condition-case nil
             (progn (json-parse-string s) t)
           (error
            ;; вывод внешней программы мог прийти «сырыми байтами» (не UTF-8-строкой):
            ;; перекодируем туда-обратно, а если и так не вышло — медленный парсер json.el
            (condition-case nil
                (progn (json-parse-string
                        (decode-coding-string (encode-coding-string s 'utf-8) 'utf-8))
                       t)
              (error
               (condition-case nil
                   (progn (require 'json) (json-read-from-string s) t)
                 (error nil))))))))

  (defun morg--js-object-string-p (s lang)
    "Похоже ли S на вывод console.log объекта/массива из блока js/ts (не JSON)."
    (and (stringp s)
         (member lang '("js" "javascript" "ts" "typescript"))
         (< (length s) morg-results-json-max)
         (string-match-p "\\`[ \t\n]*[{[]" s)
         (string-match-p "[}\\]][ \t\n]*\\'" s)))

  (defun morg--results-json-advice (args)
    "Фильтр аргументов org-babel-insert-result: JSON → «code» с языком json,
вывод console.log из js/ts (объект с ключами без кавычек) → «code» с языком js."
    (let* ((result (nth 0 args))
           (params (nth 1 args))
           (lang   (nth 4 args))
           (new-lang
            (when (and morg-results-auto-json
                       (not (seq-intersection
                             params '("silent" "none" "table" "vector" "list" "raw" "file"
                                      "link" "graphics" "html" "latex" "org" "drawer"
                                      "code" "pp" "verbatim"))))
              (cond ((morg--json-string-p result) "json")
                    ((morg--js-object-string-p result lang) "js")))))
      ;; диагностика: вывод похож на JSON, но не преобразован — сказать почему (в *Messages*)
      (when (and (not new-lang) (stringp result)
                 (string-match-p "\\`[ \t\n]*[{[]" result))
        (message "morg-json: пропущено; :results=%S lang=%S длина=%d; %s"
                 params lang (length result)
                 (cond ((not morg-results-auto-json) "morg-results-auto-json = nil")
                       ((seq-intersection params '("silent" "none" "table" "vector" "list" "raw" "file"
                                                   "link" "graphics" "html" "latex" "org" "drawer"
                                                   "code" "pp" "verbatim"))
                        "в :results есть слово, отключающее преобразование")
                       (t (condition-case err
                              (progn (json-parse-string result) "JSON корректен?!")
                            (error (format "ошибка разбора: %S" err)))))))
      (if new-lang
          (let ((a (append args (make-list (max 0 (- 5 (length args))) nil))))
            (setf (nth 1 a) (cons "code" params))
            (setf (nth 4 a) new-lang)
            a)
        args)))

  (with-eval-after-load 'ob-core
    (advice-add 'org-babel-insert-result :filter-args #'morg--results-json-advice))

  ;; Выполнение блока всегда в UTF-8: временные файлы (запрос psql, скрипт shell) пишутся и
  ;; читаются как UTF-8, поэтому Emacs не спрашивает «Select coding system», а вывод
  ;; программ с кириллицей не превращается в «сырые байты» — даже если демон запущен
  ;; без UTF-8-локали.
  (define-advice org-babel-execute-src-block (:around (fn &rest args) morg-utf8)
    (let ((coding-system-for-read  'utf-8)
          (coding-system-for-write 'utf-8))
      (apply fn args)))

  ;; --- фоновый запуск блоков и время выполнения --------------------------------------
  ;; Фоновый запуск (пакет ob-async): блок с заголовком :async выполняется в отдельном
  ;; процессе Emacs, пока вы продолжаете работать; на месте результата сначала стоит
  ;; #+RESULTS[хеш]:, потом приходит вывод. Пример:
  ;;   #+begin_src zsh :async :results output replace
  ;; Ограничения: блоки с :session не поддерживаются; запросы пароля (TRAMP /sudo::)
  ;; в фоне не работают; typescript в фоне не запускается (запуск .ts определён в этом
  ;; конфиге, а фоновый процесс его не видит) — ts-блоки быстрые, оставьте их обычными.
  (use-package ob-async
    :after org
    :custom

  (defvar morg-babel-show-time t "Показывать время выполнения src-блока.")
  (defvar morg-babel-time-min 0.3 "Не показывать, если быстрее (секунды); 0 — показывать всегда.")
  (defvar morg--babel-running nil "Очередь фоновых блоков: (МОМЕНТ-СТАРТА . ЯЗЫК).")
  (defvar morg--babel-sync nil "Не-nil, пока обычный (не фоновый) блок выполняется.")

    (when (and morg-babel-show-time (>= secs morg-babel-time-min))
      (message "⏱ %s: %s" label
               (if (< secs 1)
                   (format "%d мс" (round (* secs 1000)))
                 (format "%.2f с" secs)))))

  (defun morg--babel-timing-advice (orig &rest args)
    "Вокруг org-babel-execute-src-block: засечь время обычного блока или поставить в очередь фоновый."
    (let* ((info   (or (nth 1 args) (ignore-errors (org-babel-get-src-block-info))))
           (lang   (or (nth 0 info) "блок"))
           (params (nth 2 info))
           (async  (and (assq :async params)
                        (not (equal (cdr (assq :async params)) "no"))
                        (not (member lang (bound-and-true-p ob-async-no-async-languages-alist)))
                        (null (cdr (assq :session params)))))
           (start  (float-time)))
      (if async
          (progn (setq morg--babel-running
                       (append morg--babel-running (list (cons start lang))))
                 (apply orig args))
        (prog1 (let ((morg--babel-sync t)) (apply orig args))
          (morg--babel-report (- (float-time) start) lang)))))

  (defun morg--babel-async-done ()
    "org-babel-after-execute-hook: фоновый блок завершился — показать, сколько он шёл."
    (unless morg--babel-sync
      (when-let* ((entry (pop morg--babel-running)))
        (morg--babel-report (- (float-time) (car entry)) (concat (cdr entry) " (фон)")))))

    (advice-add 'org-babel-execute-src-block :around #'morg--babel-timing-advice
                '((depth . -100)))
    (add-hook 'org-babel-after-execute-hook #'morg--babel-async-done))

  ;; --- переход к блоку по имени (#+name:) ------------------------------------------
  ;; C-c n b — список всех именованных блоков и таблиц заметки (язык и подпись справа),
  ;; выбор переносит курсор к блоку (вернуться — C-u C-SPC). Курсор на «x=имя» в строке
  ;; #+begin_src … :var x=имя …, #+header: или #+call: и C-c C-o — прыжок к блоку-источнику.
  (defun morg--named-blocks ()
    "Список (ИМЯ ПОЗИЦИЯ ОПИСАНИЕ) для всех #+name: в буфере, в порядке появления."
    (let (res (case-fold-search t))
      (save-excursion
        (save-restriction
          (widen)
          (goto-char (point-min))
          (while (re-search-forward "^[ \t]*#\\+name:[ \t]*\\([^ \t\n]+\\)" nil t)
            (let ((name (match-string-no-properties 1))
                  (pos (line-beginning-position))
                  caption kind)
              (forward-line 1)
              (while (looking-at-p morg--affil-re)
                (when (looking-at "[ \t]*#\\+caption:[ \t]*\\(.*\\)$")
                  (setq caption (match-string-no-properties 1)))
                (forward-line 1))
              (setq kind (cond ((looking-at "[ \t]*#\\+begin_src[ \t]+\\([^ \t\n]+\\)")
                                (match-string-no-properties 1))
                               ((looking-at-p "[ \t]*|") "таблица")
                               (t "")))
              (push (list name pos (string-trim
                                    (concat kind (when (and caption (not (string-empty-p caption)))
                                                   (concat " · " caption)))))
                    res)))))
      (nreverse res)))

  (defun morg--goto-named (pos)
    (push-mark)
    (widen)
    (goto-char pos)
    (when (fboundp 'org-fold-show-context) (org-fold-show-context 'link-search))
    (recenter 3))

  (defun morg-goto-named-block ()
    "Перейти к именованному блоку/таблице (#+name:) этой заметки."
    (interactive)
    (let ((blocks (morg--named-blocks)))
      (unless blocks (user-error "В этой заметке нет #+name:"))
      (let* ((table
              (lambda (str pred action)
                (if (eq action 'metadata)
                    `(metadata
                      (display-sort-function . identity)
                      (annotation-function
                       . ,(lambda (c)
                            (let ((d (nth 2 (assoc c blocks))))
                              (and d (not (string-empty-p d)) (concat "  " d))))))
                  (complete-with-action action (mapcar #'car blocks) str pred))))
             (choice (completing-read "Блок: " table nil t)))
        (morg--goto-named (nth 1 (assoc choice blocks))))))

  (defun morg--follow-var-reference ()
    "org-open-at-point-functions: курсор на «x=имя» в заголовке блока — перейти к блоку «имя»."
    (when (save-excursion
            (beginning-of-line)
            (looking-at-p "[ \t]*#\\+\\(?:begin_src\\|headers?:\\|call:\\)"))
      (let* ((beg (save-excursion (skip-chars-backward "^ \t\n") (point)))
             (end (save-excursion (skip-chars-forward "^ \t\n") (point)))
             (tok (buffer-substring-no-properties beg end))
             (name (cond ((string-match "=\\([^ \t\n(\\[]+\\)" tok) (match-string 1 tok))
                         ((string-match "\\`\\([^ \t\n(\\[=]+\\)" tok) (match-string 1 tok))))
             (hit (and name (assoc name (morg--named-blocks)))))
        (when hit
          (morg--goto-named (nth 1 hit))
          t))))

  (defun morg--babel-result (start end alist &optional space prefix-fn)
    "Результат для capf: ALIST из (ВАРИАНТ . ОПИСАНИЕ); SPACE — добавить пробел после выбора;
PREFIX-FN — функция ВАРИАНТ -> строка, которая показывается слева (иконка)."
    (append
     (list start end
           (lambda (str pred action)
             (if (eq action 'metadata)
                 '(metadata (display-sort-function . identity))
               (complete-with-action action (mapcar #'car alist) str pred)))
           :annotation-function
           (lambda (c) (let ((d (cdr (assoc c alist)))) (when d (concat "  " d))))
           :exclusive 'no)
     (when prefix-fn
       (list :affixation-function
             (lambda (cands)           ; при affixation аннотацию нужно вернуть суффиксом
               (mapcar (lambda (c)
                         (list c (funcall prefix-fn c)
                               (let ((d (cdr (assoc c alist))))
                                 (if d (propertize (concat "  " d) 'face 'completions-annotations) ""))))
                       cands))))
     (when space
       (list :exit-function
             (lambda (_ status) (when (eq status 'finished) (insert " ")))))))

  (defun morg--babel-capf ()
    "capf для строк #+begin_src / #+header: / #+call: и ключевых слов #+…."
    (when (derived-mode-p 'org-mode)
      (let* ((pt (point))
             (bol (line-beginning-position))
             (case-fold-search t))
        (cond
         ;; <<имя>> внутри src-блока
         ((and (looking-back "<<\\([^<>\n ]*\\)" bol)
               (org-in-src-block-p))
          (morg--babel-result (match-beginning 1) pt (morg--babel-names)
                              nil #'morg--babel-name-icon))
         ;; #+call: имя
         ((looking-back "^[ \t]*#\\+call:[ \t]*\\([^ \t(\n]*\\)" bol)
          (morg--babel-result (match-beginning 1) pt (morg--babel-names)
                              nil #'morg--babel-name-icon))
         ;; #+begin_src язык
         ((looking-back "^[ \t]*#\\+begin_src[ \t]+\\([^ \t\n]*\\)" bol)
          (morg--babel-result (match-beginning 1) pt (morg--babel-languages) t
                              #'morg--babel-lang-icon))
         ;; #+ключевое-слово
         ((and (looking-back "^[ \t]*\\(#\\+[[:alpha:]_]*\\)" bol)
               (not (org-in-src-block-p)))
          (morg--babel-result (match-beginning 1) pt morg--babel-keywords t
                              #'morg--babel-icon))
         ;; заголовочные аргументы в #+begin_src язык … / #+header: …
         ((save-excursion
            (goto-char bol)
            (looking-at-p "[ \t]*#\\+\\(?:begin_src[ \t]+[^ \t\n]+\\|headers?:\\)[ \t]"))
          (let* ((start (save-excursion (skip-chars-backward "^ \t") (max (point) bol)))
                 (tok (buffer-substring-no-properties start pt))
                 (before (buffer-substring-no-properties bol start))
                 (key (when (string-match
                             "\\(?:\\`\\|[ \t]\\)\\(:[[:alnum:]_-]+\\)\\(?:[ \t]+[^: \t][^ \t]*\\)*[ \t]+\\'"
                             before)
                        (match-string 1 before))))
            (cond
             ((string-prefix-p ":" tok)           ; имя аргумента
              (morg--babel-result
               start pt
               (mapcar (lambda (a) (cons (car a) (nth 1 a))) morg--babel-header-args)
               t #'morg--babel-icon))
             ((and (equal key ":var") (string-match "=" tok))   ; x=<имя блока>
              (morg--babel-result (+ start (match-end 0)) pt (morg--babel-names)
                                   nil #'morg--babel-name-icon))
             (key                                 ; значение аргумента
              (let ((vals (nth 2 (assoc key morg--babel-header-args))))
                (when vals (morg--babel-result start pt vals)))))))))))

  ;; Работает в sql-блоках с :engine postgresql — и в самой заметке, и в буфере
  ;; редактирования (C-c '). Параметры подключения (:dbhost :dbport :dbuser

  ;;   1. при первой подсказке — один запрос со списком всех таблиц (без колонок);
  ;;   2. колонки конкретной таблицы — отдельным запросом, когда она понадобилась
  ;;      (после "таблица." / "алиас." / "схема.таблица.");

            (when (plist-get c :port) (list "-p" (format "%s" (plist-get c :port))))
            (when (plist-get c :user) (list "-U" (format "%s" (plist-get c :user))))
            (when (plist-get c :db)   (list "-d" (format "%s" (plist-get c :db))))))

  (defun morg-pg--fetch (c key sql done)
    "Асинхронно выполнить SQL; по успеху вызвать DONE со списком строк (списков полей)."
    (let ((last (gethash key morg-pg--pending)))
      (unless (and last (< (- (float-time) last) morg-pg-retry-seconds))
        (puthash key (float-time) morg-pg--pending)
        (let ((buf (generate-new-buffer " *morg-pg*" t))
              (process-environment (cons "PGCONNECT_TIMEOUT=3" process-environment)))
          (condition-case err
              (make-process
               :name "morg-pg" :buffer buf :noquery t :connection-type 'pipe
               :command (append (list morg-pg-program "-X" "-w" "-A" "-t" "-F" "\t")
                                (morg-pg--args c) (list "-c" sql))
               :sentinel
               (lambda (p _event)
                 (unless (process-live-p p)
                   (if (zerop (process-exit-status p))
                       (funcall done
                                (with-current-buffer buf
                                  (mapcar (lambda (l) (split-string l "\t"))
                                          (split-string (buffer-string) "\n" t))))
                     (message "morg-pg: %s"
                              (truncate-string-to-width
                               (string-trim (with-current-buffer buf (buffer-string)))
                               160 nil nil "…")))
                   (kill-buffer buf))))
            (error (kill-buffer buf)
                   (message "morg-pg: не удалось запустить %s (%s)"
                            morg-pg-program (error-message-string err))))))))

  (defun morg-pg--resolve (c ref)
    "REF (\"таблица\" или \"схема.таблица\") -> (схема . таблица) либо nil."
    (let ((rows (morg-pg--tables c))
          (ref (replace-regexp-in-string "\"" "" ref)))
      (if (string-match "\\`\\([^.]+\\)\\.\\([^.]+\\)\\'" ref)
          (let ((s (match-string 1 ref)) (tb (match-string 2 ref)))
            (when (cl-find-if (lambda (r) (and (string-equal-ignore-case (car r) s)
                                               (string-equal-ignore-case (cadr r) tb)))
                              rows)
              (cons s tb)))
        (let ((r (or (cl-find-if (lambda (r) (and (string-equal-ignore-case (cadr r) ref)
                                                  (equal (car r) "public")))
                                 rows)
                     (cl-find-if (lambda (r) (string-equal-ignore-case (cadr r) ref))
                                 rows))))
          (and r (cons (car r) (cadr r)))))))

  (defun morg-pg--aliases (text)
    "Алиасы из FROM/JOIN в TEXT: ((\"alias\" . \"таблица\") ...), плюс сами таблицы."
    (let (res)
      (with-temp-buffer
        (insert text)
        (goto-char (point-min))
        (let ((case-fold-search t))
          (while (re-search-forward
                  "\\b\\(?:from\\|join\\|update\\|into\\)[ \t\n]+\\([[:alnum:]_.\"]+\\)\\(?:[ \t\n]+\\(?:as[ \t\n]+\\)?\\([[:alpha:]_][[:alnum:]_]*\\)\\)?"
                  nil t)
            (let ((tbl (match-string 1)) (al (match-string 2)))
              (push (cons (downcase (replace-regexp-in-string "\"" "" tbl)) tbl) res)
              (when (and al (not (member (downcase al) morg-pg--sql-words)))
                (push (cons (downcase al) tbl) res))))))
      res))

  (defun morg-pg--candidates (c parts text)
    "Кандидаты по PARTS — сегментам перед последней точкой; TEXT — текст запроса."
    (let ((rows (morg-pg--tables c)))
      (pcase (length parts)
        (0 (append
            (mapcar (lambda (s) (morg-pg--item s "  схема" 'module))
                    (delete-dups (mapcar #'car rows)))
            (cl-loop for r in rows
                     when (equal (car r) "public")
                     collect (morg-pg--item (cadr r) (morg-pg--kind-ann (nth 2 r)) 'struct))))
        (1 (let* ((p (car parts))
                  (ref (or (cdr (assoc (downcase p) (morg-pg--aliases text))) p))
                  (res (morg-pg--resolve c ref)))
             (append
              (cl-loop for r in rows
                       when (string-equal-ignore-case (car r) p)
                       collect (morg-pg--item (cadr r) (morg-pg--kind-ann (nth 2 r)) 'struct))
              (when res (morg-pg--col-items c (car res) (cdr res))))))
        (_ (let ((res (morg-pg--resolve
                       c (concat (nth (- (length parts) 2) parts) "."
                                 (car (last parts))))))
             (when res (morg-pg--col-items c (car res) (cdr res))))))))

      (ignore-errors (morg-pg-capf))))

  (defun morg-pg-refresh ()
    "Сбросить кеш подсказок PostgreSQL для подключения текущего sql-блока."
    (interactive)
    (let* ((ctx (morg-pg--context))
           (c (and ctx (morg-pg--conn (car ctx)))))
      (unless c (user-error "Поставьте курсор в sql-блок с :engine postgresql"))
      (let ((prefix (morg-pg--conn-key c)) keys)
        (dolist (tbl (list morg-pg--cache morg-pg--pending))
          (setq keys nil)
          (maphash (lambda (k _) (when (string-prefix-p prefix k) (push k keys))) tbl)
          (dolist (k keys) (remhash k tbl)))
        (morg-pg--tables c)
        (message "morg-pg: кеш сброшен, список таблиц загружается"))))

  ;; --- ER-диаграмма схемы -------------------------------------------------------------
  ;; M-x morg-pg-er-diagram (C-c n E) в sql-блоке с :engine postgresql: читает из базы
  ;; таблицы, колонки, первичные и внешние ключи выбранной схемы и вставляет под блоком
  ;; готовый d2-блок. Выполнив его (C-c C-c), получаете картинку: таблицы-прямоугольники
  ;; со списком колонок и стрелки между связанными таблицами. Нужны установленные d2
  ;; и psql. Составные внешние ключи показываются по первой колонке; связи с таблицами
  ;; из других схем не рисуются.
  (defvar morg-pg-er-max-tables 60
    "Если таблиц больше, спросить подтверждение (иначе картинка нечитаема).")
  (defvar morg-pg-er-flags "--pad 20 --layout elk" "Ключи d2 в заголовке блока.")
  (defvar morg-pg-er-dir "~/syn/org/attachments/" "Куда складывать картинки.")

  (defun morg-pg--query-sync (c sql)
    "Выполнить SQL синхронно; вернуть список строк (списков полей) или ошибку."
    (let ((process-environment (cons "PGCONNECT_TIMEOUT=5" process-environment))
          (err (make-temp-file "morg-pg-err")))
      (unwind-protect
          (with-temp-buffer
            (let ((code (apply #'call-process morg-pg-program nil (list t err) nil
                               "-X" "-w" "-A" "-t" "-F" "\t"
                               (append (morg-pg--args c) (list "-c" sql)))))
              (unless (zerop code)
                (user-error "psql: %s"
                            (string-trim (with-temp-buffer
                                           (insert-file-contents err)
                                           (buffer-string)))))
              (mapcar (lambda (l) (split-string l "\t"))
                      (split-string (buffer-string) "\n" t))))
        (delete-file err))))

  (defun morg-pg--d2-name (s)
    "Имя в кавычках для D2 (кавычки и обратные косые экранируются)."
    (format "\"%s\"" (replace-regexp-in-string "[\"\\\\]" "\\\\\\&" s)))

  (defun morg-pg--er-neighbors (picked fks schema)
    "PICKED плюс таблицы, связанные с ними внешними ключами внутри SCHEMA."
    (let ((set (copy-sequence picked)))
      (dolist (f fks)
        (when (equal (nth 2 f) schema)
          (cond ((member (car f) picked) (cl-pushnew (nth 3 f) set :test #'equal))
                ((member (nth 3 f) picked) (cl-pushnew (car f) set :test #'equal)))))
      set))

  (defun morg-pg--er-d2 (schema cols fks tables)
    "Текст диаграммы D2.
COLS — строки (таблица колонка тип pk), FKS — (таблица колонка схема2 таблица2 колонка2),
TABLES — какие таблицы рисовать."
    (let ((fk-cols (make-hash-table :test #'equal)) edges out)
      (dolist (f fks)
        (when (and (equal (nth 2 f) schema)
                   (member (car f) tables) (member (nth 3 f) tables))
          (puthash (cons (car f) (nth 1 f)) t fk-cols)
          (push f edges)))
      (push "direction: right\n\n" out)
      (dolist (tb tables)
        (push (format "%s: {\n  shape: sql_table\n" (morg-pg--d2-name tb)) out)
        (dolist (c cols)
          (when (equal (car c) tb)
            (let* ((pk (equal (nth 3 c) "t"))
                   (fk (gethash (cons tb (nth 1 c)) fk-cols))
                   (constraint (cond ((and pk fk) " {constraint: [primary_key; foreign_key]}")
                                     (pk " {constraint: primary_key}")
                                     (fk " {constraint: foreign_key}")
                                     (t ""))))
              (push (format "  %s: %s%s\n" (morg-pg--d2-name (nth 1 c))
                            (morg-pg--d2-name (nth 2 c)) constraint)
                    out))))
        (push "}\n\n" out))
      (dolist (f (nreverse edges))
        (push (format "%s.%s -> %s.%s\n"
                      (morg-pg--d2-name (nth 0 f)) (morg-pg--d2-name (nth 1 f))
                      (morg-pg--d2-name (nth 3 f)) (morg-pg--d2-name (nth 4 f)))
              out))
      (apply #'concat (nreverse out))))

  (defun morg-pg-er-diagram ()
    "Построить ER-диаграмму (d2-блок) схемы PostgreSQL для подключения текущего sql-блока."
    (interactive)
    (unless (derived-mode-p 'org-mode)
      (user-error "Запустите в самой заметке (не в буфере редактирования блока)"))
    (let* ((ctx (or (morg-pg--context)
                    (user-error "Поставьте курсор в sql-блок с :engine postgresql")))
           (c (morg-pg--conn (car ctx)))
           (schemas (mapcar #'car
                            (morg-pg--query-sync
                             c (format "SELECT nspname FROM pg_namespace WHERE nspname NOT IN (%s) AND nspname NOT LIKE 'pg\\_toast%%' AND nspname NOT LIKE 'pg\\_temp%%' ORDER BY 1"
                                       morg-pg-system-schemas))))
           (schema (completing-read "Схема: " schemas nil t nil nil
                                    (if (member "public" schemas) "public" (car schemas))))
           (qs (morg-pg--quote schema))
           (cols (morg-pg--query-sync
                  c (format "SELECT c.relname, a.attname, format_type(a.atttypid, a.atttypmod), COALESCE((SELECT true FROM pg_index i WHERE i.indrelid = c.oid AND i.indisprimary AND a.attnum = ANY(i.indkey::int2[])), false) FROM pg_attribute a JOIN pg_class c ON c.oid = a.attrelid JOIN pg_namespace n ON n.oid = c.relnamespace WHERE n.nspname = '%s' AND c.relkind IN ('r','p') AND a.attnum > 0 AND NOT a.attisdropped ORDER BY c.relname, a.attnum"
                            qs)))
           (fks (morg-pg--query-sync
                 c (format "SELECT c1.relname, a1.attname, n2.nspname, c2.relname, a2.attname FROM pg_constraint k JOIN pg_class c1 ON c1.oid = k.conrelid JOIN pg_namespace n1 ON n1.oid = c1.relnamespace JOIN pg_class c2 ON c2.oid = k.confrelid JOIN pg_namespace n2 ON n2.oid = c2.relnamespace JOIN pg_attribute a1 ON a1.attrelid = k.conrelid AND a1.attnum = k.conkey[1] JOIN pg_attribute a2 ON a2.attrelid = k.confrelid AND a2.attnum = k.confkey[1] WHERE k.contype = 'f' AND n1.nspname = '%s' ORDER BY 1,2"
                           qs)))
           (all (delete-dups (mapcar #'car cols)))
           (picked (delete "" (completing-read-multiple
                               "Таблицы через запятую (пусто — вся схема): " all)))
           (tables (sort (if picked (morg-pg--er-neighbors picked fks schema) (copy-sequence all))
                         #'string<)))
      (unless tables (user-error "В схеме %s нет таблиц" schema))
      (when (and (> (length tables) morg-pg-er-max-tables)
                 (not (y-or-n-p (format "Таблиц %d — картинка будет огромной. Продолжить? "
                                        (length tables)))))
        (user-error "Отменено: выберите нужные таблицы или поднимите morg-pg-er-max-tables"))
      (let* ((safe (lambda (s) (replace-regexp-in-string "[^[:alnum:]_-]" "_" (format "%s" s))))
             (base (format "er-%s-%s" (funcall safe (or (plist-get c :db) "db")) (funcall safe schema)))
             (file (concat (file-name-as-directory morg-pg-er-dir) base ".png"))
             (d2 (morg-pg--er-d2 schema cols fks tables)))
        (make-directory (file-name-directory (expand-file-name file)) t)
        (when (and (derived-mode-p 'org-mode) (org-in-src-block-p))
          (goto-char (org-element-property :end (org-element-at-point))))
        (unless (bolp) (insert "\n"))
        (insert (format "\n#+name: %s\n#+caption: ER-диаграмма схемы %s (%d таблиц)\n#+begin_src d2 :file %s :results file replace :flags \"%s\"\n%s#+end_src\n"
                        base schema (length tables) file morg-pg-er-flags d2))
        (re-search-backward "^[ \t]*#\\+begin_src d2" nil t)
        (message "Блок с диаграммой вставлен (%d таблиц)." (length tables))
        (cond
         ((not (executable-find "d2"))
          (message "Блок вставлен, но программа d2 не найдена в PATH — картинку не построить."))
         ((y-or-n-p "Построить картинку сейчас? ")
          (org-babel-execute-src-block)
          ;; картинку показываем чуть позже: файл уже записан, блок перерисован
          (let ((buf (current-buffer)))
            (run-with-timer
             0.3 nil
             (lambda ()
               (when (buffer-live-p buf)
                 (with-current-buffer buf
                   (condition-case err
                       (cond ((fboundp 'org-link-preview-refresh) (org-link-preview-refresh))
                             (t (org-redisplay-inline-images)))
                     (error (message "Не удалось показать картинку: %s"
                                     (error-message-string err))))))))))))))
